# Todo List Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add a linked to-do panel that expands the window rightward (420→740px), where tasks log pomodoro time automatically and support infinite subtask nesting with checkbox-based % completion.

**Architecture:** Six focused tasks: IPC resize wiring → pure logic with TDD → HTML markup → CSS layout → app.js integration → todo.js DOM wiring. `loadTasks`/`saveTasks` live in `app.js` (not `todo.js` as the spec states) because `app.js` calls `render()` at page load before `todo.js` is defined; `todo.js` uses them as globals.

**Tech Stack:** Electron IPC (win.setSize), vanilla JS ES5 (`var`), localStorage, Jest 29, CSS transform/opacity transitions.

**Spec deviation note:** The spec places `loadTasks`/`saveTasks` in `todo.js`, but since `app.js` loads first and calls `render()` immediately (which needs `findTask` and `loadTasks`), these helpers must be defined in `app.js`. `todo.js` calls them as globals.

```javascript
// ── ID generation ───────────────────────────────

function generateId() {
  if (typeof crypto !== 'undefined' && crypto.randomUUID) {
    return crypto.randomUUID()
  }
  return Math.random().toString(36).slice(2) + Date.now().toString(36)
}

// ── Task creation ───────────────────────────────

function createTask(name) {
  return {
    id: generateId(),
    name: name,
    checked: false,
    estimatedMinutes: null,
    loggedMinutes: 0,
    children: [],
    collapsed: false
  }
}

// ── Internal helpers ────────────────────────────

function setCheckedDeep(node, checked) {
  return Object.assign({}, node, {
    checked: checked,
    children: node.children.map(function (c) { return setCheckedDeep(c, checked) })
  })
}

function mapTree(tasks, fn) {
  return tasks.map(function (task) {
    var updated = fn(task)
    return Object.assign({}, updated, {
      children: mapTree(updated.children, fn)
    })
  })
}

function checkTask(tasks, id, checked) {
  function processNode(node) {
    if (node.id === id) {
      return setCheckedDeep(node, checked)
    }
    var newChildren = checkTask(node.children, id, checked)
    var allChecked = newChildren.length > 0 && newChildren.every(function (c) { return c.checked })
    var anyChecked = newChildren.some(function (c) { return c.checked })
    var parentChecked = newChildren.length > 0
      ? (checked ? allChecked : (node.checked && anyChecked ? allChecked : false))
      : node.checked
    return Object.assign({}, node, {
      children: newChildren,
      checked: parentChecked
    })
  }
  return tasks.map(processNode)
}

// ── Completion calculation ──────────────────────

function getLeafCount(node) {
  if (node.children.length === 0) return 1
  return node.children.reduce(function (sum, c) { return sum + getLeafCount(c) }, 0)
}

function getCheckedLeafCount(node) {
  if (node.children.length === 0) return node.checked ? 1 : 0
  return node.children.reduce(function (sum, c) { return sum + getCheckedLeafCount(c) }, 0)
}

function getCompletionPercent(node) {
  var total = getLeafCount(node)
  if (total === 0) return 0
  return Math.round((getCheckedLeafCount(node) / total) * 100)
}

function getOverallPercent(tasks) {
  if (tasks.length === 0) return 0
  var totalLeaves = tasks.reduce(function (sum, t) { return sum + getLeafCount(t) }, 0)
  if (totalLeaves === 0) return 0
  var checkedLeaves = tasks.reduce(function (sum, t) { return sum + getCheckedLeafCount(t) }, 0)
  return Math.round((checkedLeaves / totalLeaves) * 100)
}

// ── Time logging ────────────────────────────────

function addMinutes(tasks, id, minutes) {
  return tasks.map(function (task) {
    if (task.id === id) {
      return Object.assign({}, task, { loggedMinutes: task.loggedMinutes + minutes })
    }
    return Object.assign({}, task, { children: addMinutes(task.children, id, minutes) })
  })
}

// ── Task CRUD ───────────────────────────────────

function addTask(tasks, parentId, task) {
  if (parentId === null) {
    return tasks.concat([task])
  }
  return tasks.map(function (t) {
    if (t.id === parentId) {
      return Object.assign({}, t, { children: t.children.concat([task]) })
    }
    return Object.assign({}, t, { children: addTask(t.children, parentId, task) })
  })
}

function deleteTask(tasks, id) {
  return tasks
    .filter(function (t) { return t.id !== id })
    .map(function (t) {
      return Object.assign({}, t, { children: deleteTask(t.children, id) })
    })
}

function findTask(tasks, id) {
  for (var i = 0; i < tasks.length; i++) {
    if (tasks[i].id === id) return tasks[i]
    var found = findTask(tasks[i].children, id)
    if (found) return found
  }
  return null
}

// ── Input parsing / formatting ──────────────────

function parseTimeInput(str) {
  if (!str) return null
  str = str.trim()
  var hourMatch = str.match(/^(\d+(?:\.\d+)?)h$/)
  if (hourMatch) {
    var mins = Math.round(parseFloat(hourMatch[1]) * 60)
    return mins > 0 ? mins : null
  }
  var minMatch = str.match(/^(\d+)m$/)
  if (minMatch) {
    var m = parseInt(minMatch[1], 10)
    return m > 0 ? m : null
  }
  var numMatch = str.match(/^(\d+)$/)
  if (numMatch) {
    var n = parseInt(numMatch[1], 10)
    return n > 0 ? n : null
  }
  return null
}

function formatLoggedTime(minutes) {
  var h = Math.floor(minutes / 60)
  var m = minutes % 60
  return h + ':' + (m < 10 ? '0' : '') + m
}

// ── UMD-lite export ─────────────────────────────

if (typeof module !== 'undefined' && module.exports) {
  module.exports = {
    createTask: createTask,
    checkTask: checkTask,
    getLeafCount: getLeafCount,
    getCheckedLeafCount: getCheckedLeafCount,
    getCompletionPercent: getCompletionPercent,
    getOverallPercent: getOverallPercent,
    addMinutes: addMinutes,
    addTask: addTask,
    deleteTask: deleteTask,
    findTask: findTask,
    parseTimeInput: parseTimeInput,
    formatLoggedTime: formatLoggedTime
  }
}
```

function escapeHtml(str) {
  return str
    .replace(/&/g, '&amp;')
    .replace(/</g, '&lt;')
    .replace(/>/g, '&gt;')
    .replace(/"/g, '&quot;')
}

function formatEstimate(minutes) {
  if (!minutes) return '\u2014'
  if (minutes % 60 === 0) return (minutes / 60) + 'h'
  var h = Math.floor(minutes / 60)
  var m = minutes % 60
  return h > 0 ? h + 'h ' + m + 'm' : m + 'm'
}

// ── Inline name editing ─────────────────────────

function startInlineNameEdit(nameEl, taskId, isNew) {
  var current = nameEl.textContent
  var input = document.createElement('input')
  input.type = 'text'
  input.className = 'task-name-input'
  input.value = isNew ? '' : current
  nameEl.replaceWith(input)
  input.focus()

  function save() {
    var val = input.value.trim()
    if (!val) {
      if (isNew) {
        var tasks = loadTasks()
        tasks = deleteTask(tasks, taskId)
        saveTasks(tasks)
      }
      renderTodoPanel()
      return
    }
    var tasks = loadTasks()
    tasks = updateTaskName(tasks, taskId, val)
    saveTasks(tasks)
    renderTodoPanel()
  }

  input.addEventListener('keydown', function (e) {
    if (e.key === 'Enter') { e.preventDefault(); save() }
    if (e.key === 'Escape') {
      if (isNew) {
        var tasks = loadTasks()
        tasks = deleteTask(tasks, taskId)
        saveTasks(tasks)
      }
      renderTodoPanel()
    }
  })
  input.addEventListener('blur', save)
}

// ── Update task name in tree ────────────────────

function updateTaskName(tasks, id, name) {
  return tasks.map(function (t) {
    if (t.id === id) return Object.assign({}, t, { name: name })
    return Object.assign({}, t, { children: updateTaskName(t.children, id, name) })
  })
}

// ── Update task collapsed state ─────────────────

function setCollapsed(tasks, id, collapsed) {
  return tasks.map(function (t) {
    if (t.id === id) return Object.assign({}, t, { collapsed: collapsed })
    return Object.assign({}, t, { children: setCollapsed(t.children, id, collapsed) })
  })
}

// ── Inline estimate editing ─────────────────────

function startInlineEstEdit(estSpan, taskId) {
  var input = document.createElement('input')
  input.type = 'text'
  input.className = 'task-est-input'
  input.placeholder = '2h / 90m'
  estSpan.replaceWith(input)
  input.focus()

  function save() {
    var val = input.value.trim()
    var minutes = val ? parseTimeInput(val) : null
    var tasks = loadTasks()
    tasks = tasks.map(function updateEst(t) {
      if (t.id === taskId) return Object.assign({}, t, { estimatedMinutes: minutes })
      return Object.assign({}, t, { children: t.children.map(updateEst) })
    })
    saveTasks(tasks)
    renderTodoPanel()
  }

  input.addEventListener('keydown', function (e) {
    if (e.key === 'Enter') { e.preventDefault(); save() }
    if (e.key === 'Escape') renderTodoPanel()
  })
  input.addEventListener('blur', save)
}

// ── Build task element ──────────────────────────

function buildTaskEl(task, depth) {
  var row = document.createElement('div')
  row.className = 'task-row'

  var main = document.createElement('div')
  main.className = 'task-row-main' + (state.activeTaskId === task.id ? ' active-task' : '')

  // Indent
  if (depth > 0) {
    var indent = document.createElement('div')
    indent.className = 'task-indent'
    indent.style.width = (depth * 14) + 'px'
    main.appendChild(indent)
  }

  // Collapse toggle or spacer
  if (task.children.length > 0) {
    var colBtn = document.createElement('button')
    colBtn.className = 'task-collapse-btn'
    colBtn.textContent = task.collapsed ? '\u25B6' : '\u25BC'
    colBtn.title = task.collapsed ? 'Expand' : 'Collapse'
    colBtn.addEventListener('click', function () {
      var tasks = loadTasks()
      tasks = setCollapsed(tasks, task.id, !task.collapsed)
      saveTasks(tasks)
      renderTodoPanel()
    })
    main.appendChild(colBtn)
  } else {
    var spacer = document.createElement('div')
    spacer.className = 'task-collapse-spacer'
    main.appendChild(spacer)
  }

  // Checkbox
  var checkbox = document.createElement('input')
  checkbox.type = 'checkbox'
  checkbox.className = 'task-checkbox'
  checkbox.checked = task.checked
  checkbox.addEventListener('change', function () {
    var tasks = loadTasks()
    tasks = checkTask(tasks, task.id, checkbox.checked)
    saveTasks(tasks)
    renderTodoPanel()
  })
  main.appendChild(checkbox)

  // Task name
  var nameEl = document.createElement('span')
  nameEl.className = 'task-name' + (task.checked ? ' checked' : '')
  nameEl.textContent = task.name || '(unnamed)'
  nameEl.addEventListener('click', function () {
    startInlineNameEdit(nameEl, task.id, false)
  })
  main.appendChild(nameEl)

  // Add child button
  var addChildBtn = document.createElement('button')
  addChildBtn.className = 'task-add-child-btn'
  addChildBtn.textContent = '+'
  addChildBtn.title = 'Add subtask'
  addChildBtn.addEventListener('click', function () {

    tasks = addTask(tasks, task.id, newTask)
    if (task.collapsed) tasks = setCollapsed(tasks, task.id, false)
    saveTasks(tasks)
    renderTodoPanel()
    // Focus the new task's name input
    var newEl = elTodoList.querySelector('[data-task-id="' + newTask.id + '"] .task-name')
    if (newEl) startInlineNameEdit(newEl, newTask.id, true)
  })
  main.appendChild(addChildBtn)

  // Select button
  var selectBtn = document.createElement('button')
  selectBtn.className = 'task-select-btn' + (state.activeTaskId === task.id ? ' selected' : '')
  selectBtn.textContent = state.activeTaskId === task.id ? 'Active' : 'Select'
  selectBtn.addEventListener('click', function () {
    state.activeTaskId = state.activeTaskId === task.id ? null : task.id
    render()
    renderTodoPanel()
  })
  main.appendChild(selectBtn)

  // Delete button
  var delBtn = document.createElement('button')
  delBtn.className = 'task-delete-btn'
  delBtn.textContent = '\u00D7'
  delBtn.title = 'Delete task'

      render()
    }

    saveTasks(tasks)

  row.appendChild(main)

  // Meta row (progress bar + time)
